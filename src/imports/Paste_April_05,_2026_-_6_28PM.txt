    <script>
        // ==========================================
        // 1. 초기 설정 및 유틸리티
        // ==========================================
        const cursor = document.getElementById('cursor');
        const hudClock = document.getElementById('hud-clock');
        
        // 커서 이동
        window.addEventListener('mousemove', (e) => {
            cursor.style.left = e.clientX + 'px';
            cursor.style.top = e.clientY + 'px';
        });

        // 인터랙티브 요소 커서 호버 효과 (항목 전체 크기에 반응하도록 타겟 변경)
        const interativeElements = document.querySelectorAll('button, .contact-item, input, textarea');
        interativeElements.forEach(el => {
            el.addEventListener('mouseenter', () => cursor.classList.add('hover'));
            el.addEventListener('mouseleave', () => cursor.classList.remove('hover'));
        });

        // 실시간 HUD 시계
        setInterval(() => {
            const now = new Date();
            hudClock.innerText = now.toLocaleTimeString('en-US', { hour12: false }) + ':' + now.getMilliseconds().toString().padStart(3, '0').slice(0, 2);
        }, 50);

        // 텍스처 생성을 위한 유틸리티 함수 (외부 이미지 대신 다채로운 캔버스 텍스처 생성)
        function createProceduralTexture(index) {
            const canvas = document.createElement('canvas');
            canvas.width = 512;
            canvas.height = 640; // 4:5 비율로 높이 수정 (기존 768)
            const ctx = canvas.getContext('2d');
            
            // 다이내믹 그라데이션
            const hue1 = (index * 45) % 360;
            const hue2 = (hue1 + 60) % 360;
            const gradient = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            gradient.addColorStop(0, `hsl(${hue1}, 80%, 30%)`);
            gradient.addColorStop(1, `hsl(${hue2}, 80%, 10%)`);
            
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // 타이포그래피 장식
            ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
            ctx.font = 'bold 60px "Space Grotesk", sans-serif';
            ctx.fillText(`PROJECT`, 40, 100);
            ctx.fillText(`#${index + 1}`, 40, 160);

            // 기하학적 도형 (비디오 재생을 시뮬레이션하기 위한 장치)
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(canvas.width/2, canvas.height/2, 100, 0, Math.PI * 2);
            ctx.stroke();

            ctx.font = '20px "Space Mono", monospace';
            ctx.fillText(`ID: ${Math.random().toString(36).substr(2, 6).toUpperCase()}`, 40, canvas.height - 40);

            const texture = new THREE.CanvasTexture(canvas);
            texture.needsUpdate = true;
            return { texture, canvas }; // 캔버스 참조도 반환하여 나중에 수정 가능하도록 함
        }

        // ==========================================
        // 2. Three.js 3D 갤러리 셋업
        // ==========================================
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.fog = new THREE.FogExp2(0x050505, 0.02); // 깊이감을 위한 안개 (우주 공간에 맞게 밀도 조정)

        const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 0, 0); // 사용자는 공간 중앙에 위치

        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        container.appendChild(renderer.domElement);

        const galleryGroup = new THREE.Group();
        scene.add(galleryGroup);

        // 배경 별(Stars) 파티클 추가하여 우주 공간 느낌 극대화
        const starsGeometry = new THREE.BufferGeometry();
        const starsVertices = [];
        for(let i=0; i<1500; i++) {
            const x = (Math.random() - 0.5) * 300;
            const y = (Math.random() - 0.5) * 300;
            const z = (Math.random() - 0.5) * 300;
            starsVertices.push(x, y, z);
        }
        starsGeometry.setAttribute('position', new THREE.Float32BufferAttribute(starsVertices, 3));
        const starsMaterial = new THREE.PointsMaterial({color: 0xffffff, size: 0.2, transparent: true, opacity: 0.6});
        const starField = new THREE.Points(starsGeometry, starsMaterial);
        scene.add(starField);

        // 우주 공간 플로팅 레이아웃 변수
        const totalItems = 60; // 아이템 개수 증가 (풍성한 공간감)
        const itemWidth = 4;
        const itemHeight = 5; 
        const planes = [];

        // 갤러리 아이템 생성
        const geometry = new THREE.PlaneGeometry(itemWidth, itemHeight, 32, 32);
        
        for (let i = 0; i < totalItems; i++) {
            const { texture, canvas } = createProceduralTexture(i);
            
            const material = new THREE.MeshBasicMaterial({ 
                map: texture,
                side: THREE.DoubleSide,
                transparent: true,
                opacity: 0.8
            });

            const mesh = new THREE.Mesh(geometry, material);
            
            // 피보나치 구면(Fibonacci Sphere) 알고리즘을 이용한 겹침 없는 3D 구형 배치
            const phi = Math.acos(1 - 2 * (i + 0.5) / totalItems);
            const theta = Math.PI * (1 + Math.sqrt(5)) * i;

            // 중앙에서 일정 거리 이상 떨어지도록 반경(Radius)에 랜덤 깊이감 추가
            const radius = 20 + Math.random() * 30; 

            mesh.position.x = radius * Math.sin(phi) * Math.cos(theta);
            mesh.position.y = radius * Math.cos(phi);
            mesh.position.z = radius * Math.sin(phi) * Math.sin(theta);
            
            // 모든 이미지가 우주 중앙(카메라 시작점)을 바라보게 하여 둥근 공간감을 만듦
            mesh.lookAt(0, 0, 0);

            // 사용자 데이터 저장 (상호작용 용도)
            mesh.userData = { 
                id: i, 
                originalScale: 1, 
                canvas: canvas, 
                baseTexture: texture 
            };
            
            galleryGroup.add(mesh);
            planes.push(mesh);
        }

        // ==========================================
        // 3. 물리 기반 관성 스크롤 및 드래그 (Interaction)
        // ==========================================
        let isDragging = false;
        let startX = 0, startY = 0;
        let currentX = 0, currentY = 0;
        
        // 회전 타겟 값 (우주 공간이므로 X축, Y축 모두 회전)
        let targetRotationX = 0;
        let targetRotationY = 0;
        
        // 관성을 위한 속도 변수
        let velocityX = 0;
        let velocityY = 0;

        // 마우스/터치 이벤트
        const onPointerDown = (e) => {
            if(document.querySelector('#page-detail').classList.contains('active')) return;
            isDragging = true;
            startX = e.clientX || e.touches[0].clientX;
            startY = e.clientY || e.touches[0].clientY;
            cursor.style.transform = 'translate(-50%, -50%) scale(0.5)';
        };

        const onPointerMove = (e) => {
            if (!isDragging) return;
            const clientX = e.clientX || e.touches[0].clientX;
            const clientY = e.clientY || e.touches[0].clientY;
            
            const deltaX = clientX - startX;
            const deltaY = clientY - startY;

            // X축, Y축 드래그에 따라 공간 전체가 회전하도록 속도 축적
            velocityX += deltaX * 0.0002;
            velocityY += deltaY * 0.0002;

            startX = clientX;
            startY = clientY;
        };

        const onPointerUp = () => {
            isDragging = false;
            cursor.style.transform = 'translate(-50%, -50%) scale(1)';
        };

        window.addEventListener('mousedown', onPointerDown);
        window.addEventListener('mousemove', onPointerMove);
        window.addEventListener('mouseup', onPointerUp);
        window.addEventListener('touchstart', onPointerDown);
        window.addEventListener('touchmove', onPointerMove);
        window.addEventListener('touchend', onPointerUp);

        // 휠 스크롤로 상하 이동 보조
        window.addEventListener('wheel', (e) => {
            if(document.querySelector('#page-detail').classList.contains('active')) return;
            velocityY -= e.deltaY * 0.0001; // 휠 스크롤도 우주 공간의 X축 회전으로 반영
        });

        // ==========================================
        // 4. Raycaster (Hover 마이크로 인터랙션 및 Click)
        // ==========================================
        const raycaster = new THREE.Raycaster();
        const mouse = new THREE.Vector2();
        let hoveredMesh = null;

        window.addEventListener('mousemove', (e) => {
            // 드래그 중이거나 상세 페이지가 열려있을 때는 호버 인식 중단
            if(isDragging || document.getElementById('page-detail').classList.contains('active')) return;
            
            mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects(planes);

            if (intersects.length > 0) {
                const object = intersects[0].object;
                if (hoveredMesh !== object) {
                    // 이전 호버 해제
                    if (hoveredMesh) {
                        gsap.to(hoveredMesh.scale, { x: 1, y: 1, z: 1, duration: 0.4, ease: "power2.out" });
                        hoveredMesh.material.opacity = 0.8;
                    }
                    // 새 호버 적용 (크기 증가 및 밝기 증가 - 미디어 재생 시뮬레이션)
                    hoveredMesh = object;
                    gsap.to(hoveredMesh.scale, { x: 1.1, y: 1.1, z: 1.1, duration: 0.4, ease: "back.out(1.5)" });
                    hoveredMesh.material.opacity = 1.0;
                    cursor.classList.add('hover');
                }
            } else {
                if (hoveredMesh) {
                    gsap.to(hoveredMesh.scale, { x: 1, y: 1, z: 1, duration: 0.4, ease: "power2.out" });
                    hoveredMesh.material.opacity = 0.8;
                    hoveredMesh = null;
                    cursor.classList.remove('hover');
                }
            }
        });

        function expandProject(mesh) {
            // 1. 카메라를 해당 메쉬 쪽으로 줌인 (공간감 있는 트랜지션)
            // 메쉬의 현재 전역 위치 파악
            const targetPos = new THREE.Vector3();
            mesh.getWorldPosition(targetPos);
            
            // 카메라 최종 목적지: 메쉬 앞(중심 방향)으로 일정 거리(8) 띄운 위치
            const camTarget = targetPos.clone().normalize().multiplyScalar(targetPos.length() - 8);

            // 카메라가 현재 바라보는 방향과 목표 방향 벡터
            const startLookAt = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion).normalize();
            const endLookAt = targetPos.clone().normalize();
            
            const lookAtProxy = { t: 0 };

            // GSAP를 이용한 카메라 위치 이동
            gsap.to(camera.position, {
                x: camTarget.x,
                y: camTarget.y,
                z: camTarget.z,
                duration: 1.2,
                ease: "power3.inOut"
            });
            
            // GSAP를 이용한 카메라 시선(lookAt) 회전 보간
            gsap.to(lookAtProxy, {
                t: 1,
                duration: 1.2,
                ease: "power3.inOut",
                onUpdate: () => {
                    // 카메라가 부드럽게 메쉬를 향해 고개를 돌림
                    const currentDir = new THREE.Vector3().lerpVectors(startLookAt, endLookAt, lookAtProxy.t).normalize();
                    const lookAtTarget = camera.position.clone().add(currentDir);
                    camera.lookAt(lookAtTarget);
                }
            });

            gsap.to(camera, {
                fov: 40, // 줌인 시 시야각을 살짝 좁힘
                duration: 1.2,
                ease: "power3.inOut",
                onUpdate: () => camera.updateProjectionMatrix(),
                onComplete: () => {
                    // 2. HTML 상세 페이지 표시 (Seamless)
                    const detailPage = document.getElementById('page-detail');
                    const closeBtn = document.getElementById('close-detail');
                    const detailHero = document.getElementById('detail-hero-img');
                    
                    // 캔버스의 데이터를 CSS 배경으로 변환하여 자연스럽게 연결
                    const dataUrl = mesh.userData.canvas.toDataURL();
                    detailHero.style.backgroundImage = `url(${dataUrl})`;
                    document.getElementById('detail-title').innerText = `PROJECT #${mesh.userData.id + 1}`;

                    detailPage.classList.add('active');
                    closeBtn.style.display = 'block';
                    
                    gsap.fromTo(detailHero, { scale: 1.1 }, { scale: 1, duration: 0.8, ease: "power2.out" });
                    gsap.to(detailPage, { opacity: 1, duration: 0.5 });
                    gsap.to(closeBtn, { opacity: 1, duration: 0.5 });
                    
                    // WebGL 캔버스 서서히 페이드 아웃
                    gsap.to(container, { opacity: 0, duration: 0.5 });
                }
            });
        }

            const startLookAt = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion).normalize();

        function switchPage(target) {
            // 활성화된 HTML 페이지 숨기기
            Object.values(pages).forEach(page => {
                if(page && page.classList.contains('active')) {
                    gsap.to(page, { opacity: 0, duration: 0.4, onComplete: () => page.classList.remove('active') });

                gsap.to(container, { opacity: 1, duration: 1, ease: "power2.out" });
                gsap.to(camera.position, { x: 0, y: 0, z: 0, duration: 1, ease: "power2.out" });
            } else {
                gsap.to(container, { opacity: 0.2, duration: 0.8 });
                gsap.to(camera.position, { z: 30, duration: 0.8, ease: "power2.inOut" });

                const targetPage = pages[target];

                // 페이지별 진입 애니메이션
                if(target === 'about') {
                    gsap.fromTo(targetPage.children, { y: 50, opacity: 0 }, { y: 0, opacity: 1, duration: 0.8, stagger: 0.2, ease: "power3.out", delay: 0.3 });
                } else if (target === 'contact') {
                    gsap.fromTo('.contact-item', { x: -50, opacity: 0 }, { x: 0, opacity: 1, duration: 0.8, stagger: 0.1, ease: "power3.out", delay: 0.3 });
                } else if (target === 'mail') {

            }

        function animate() {
            requestAnimationFrame(animate);

            // 관성 물리 연산 (마찰력 적용)
            velocityX *= 0.95; 
            velocityY *= 0.95;

            // 속도를 목표값에 더함
            targetRotationY += velocityX;
            targetRotationX += velocityY;

            // 부드러운 보간 (Lerp)을 통해 우주 공간 전체 그룹 회전 적용
            galleryGroup.rotation.y = THREE.MathUtils.lerp(galleryGroup.rotation.y, targetRotationY, 0.1);
            galleryGroup.rotation.x = THREE.MathUtils.lerp(galleryGroup.rotation.x, targetRotationX, 0.1);

            // 별똥별처럼 우주 배경을 아주 천천히 회전시켜 생동감 부여
            starField.rotation.y += 0.0005;
            starField.rotation.x += 0.0002;

            renderer.render(scene, camera);
        }
